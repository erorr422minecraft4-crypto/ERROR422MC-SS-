local function getCharacter()
	local character = player.Character
	if not character then return nil end
	local humanoid = character:FindFirstChildOfClass("Humanoid")
	local root = character:FindFirstChild("HumanoidRootPart")
	if not humanoid or not root then return nil end
	return character, humanoid, root
end

local function findPlayerByName(name)
	if not name or name == "" then return nil end
	name = name:lower()

	for _, plr in ipairs(Players:GetPlayers()) do
		if plr.Name:lower() == name then
			return plr
		end
	end

	for _, plr in ipairs(Players:GetPlayers()) do
		if plr.Name:lower():sub(1, #name) == name then
			return plr
		end
	end

	return nil
end

local flying = false
local flySpeed = 75
local flyGyro, flyVelocity, flyConn

local function startFly()
	if flying then return end
	local _, humanoid, root = getCharacter()
	if not humanoid or not root then return end

	flying = true

	flyGyro = Instance.new("BodyGyro")
	flyGyro.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
	flyGyro.P = 100000
	flyGyro.D = 1000
	flyGyro.CFrame = root.CFrame
	flyGyro.Parent = root

	flyVelocity = Instance.new("BodyVelocity")
	flyVelocity.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
	flyVelocity.P = 100000
	flyVelocity.Velocity = Vector3.zero
	flyVelocity.Parent = root

	humanoid.PlatformStand = true
	humanoid.AutoRotate = false

	flyConn = RS.RenderStepped:Connect(function()
		if not flying then return end
		local _, h, r = getCharacter()
		if not h or not r or not flyGyro or not flyVelocity then return end

		flyGyro.CFrame = CFrame.lookAt(r.Position, r.Position + camera.CFrame.LookVector)

		local move = h.MoveDirection
		if move.Magnitude > 0 then
			local forward = camera.CFrame.LookVector
			local right = camera.CFrame.RightVector

			local forwardAmount = move:Dot(Vector3.new(forward.X, 0, forward.Z))
			local rightAmount = move:Dot(Vector3.new(right.X, 0, right.Z))

			local direction
			if math.abs(forwardAmount) > math.abs(rightAmount) then
				direction = forward * math.sign(forwardAmount)
			else
				direction = right * math.sign(rightAmount)
			end

			if direction.Magnitude > 0 then
				flyVelocity.Velocity = direction.Unit * flySpeed
			end
		else
			flyVelocity.Velocity = Vector3.zero
		end
	end)
end

local function stopFly()
	if not flying then return end
	flying = false

	if flyConn then flyConn:Disconnect() flyConn = nil end
	if flyGyro then flyGyro:Destroy() flyGyro = nil end
	if flyVelocity then flyVelocity:Destroy() flyVelocity = nil end

	local _, humanoid = getCharacter()
	if humanoid then
		humanoid.PlatformStand = false
		humanoid.AutoRotate = true
	end
end

local function setFlySpeed(value)
	local n = tonumber(value)
	if n then flySpeed = math.clamp(n, 1, 999) end
end

local noclipping = false

local function startNoclip()
	if noclipping then return end
	noclipping = true

	noclipConn = RS.Stepped:Connect(function()
		local character = player.Character
		if not character then return end

		end

	if not noclipping then return end
	noclipping = false

	if noclipConn then
		noclipConn:Disconnect()
		noclipConn = nil
	end

	local character = player.Character
	if character then
		for _, part in ipairs(character:GetDescendants()) do
			if part:IsA("BasePart") then
				part.CanCollide = true
			end
		end
	end
end

local frozen = false
local savedTransforms = {}
local savedWalkSpeed, savedJumpPower, savedAutoRotate = DEFAULT_WALKSPEED, DEFAULT_JUMPPOWER, true

local function freeze()
	if frozen then return end
	local character, humanoid, root = getCharacter()
	if not humanoid or not root then return end

	frozen = true
	savedTransforms = {}
	savedWalkSpeed = humanoid.WalkSpeed
	savedJumpPower = humanoid.JumpPower
	savedAutoRotate = humanoid.AutoRotate

	for _, obj in ipairs(character:GetDescendants()) do
		if obj:IsA("Motor6D") then
			savedTransforms[obj] = obj.Transform
		end
	end

	for _, track in ipairs(humanoid:GetPlayingAnimationTracks()) do
		track:Stop(0)
	end

	humanoid.WalkSpeed = 0
	humanoid.JumpPower = 0
	humanoid.AutoRotate = false
	root.Anchored = true

	for motor, transform in pairs(savedTransforms) do
		if motor.Parent then
			motor.Transform = transform
		end
	end
end

local function unfreeze()
	if not frozen then return end
	frozen = false

	local _, humanoid, root = getCharacter()
	if root then root.Anchored = false end
	if humanoid then
		humanoid.WalkSpeed = savedWalkSpeed
		humanoid.JumpPower = savedJumpPower
		humanoid.AutoRotate = savedAutoRotate
	end

	savedTransforms = {}
end

local freecamEnabled = false
local freecamKeys = {}
local freecamConn, freecamInputBegan, freecamInputEnded, freecamInputChanged
local freecamSpeed = 50
local freecamSens = 0.25
local freecamYaw, freecamPitch, freecamPos
local freecamOldType, freecamOldSubject

local function startFreecam()
	if freecamEnabled then return end
	freecamEnabled = true
	freecamKeys = {}

	freecamOldType = camera.CameraType
	freecamOldSubject = camera.CameraSubject
	freecamPos = camera.CFrame.Position

	local rx, ry = camera.CFrame:ToOrientation()
	freecamPitch = math.deg(rx)
	freecamYaw = math.deg(ry)

	camera.CameraType = Enum.CameraType.Scriptable
	UIS.MouseBehavior = Enum.MouseBehavior.LockCenter
	UIS.MouseIconEnabled = false

	freeze()

	freecamInputBegan = UIS.InputBegan:Connect(function(input, processed)
		if processed then return end
		local k = input.KeyCode
		if k == Enum.KeyCode.W then freecamKeys.W = true end
		if k == Enum.KeyCode.S then freecamKeys.S = true end
		if k == Enum.KeyCode.A then freecamKeys.A = true end
		if k == Enum.KeyCode.D then freecamKeys.D = true end
		if k == Enum.KeyCode.Space then freecamKeys.Space = true end
		if k == Enum.KeyCode.LeftControl then freecamKeys.Ctrl = true end
	end)

	freecamInputEnded = UIS.InputEnded:Connect(function(input)
		local k = input.KeyCode
		if k == Enum.KeyCode.W then freecamKeys.W = false end
		if k == Enum.KeyCode.S then freecamKeys.S = false end
		if k == Enum.KeyCode.A then freecamKeys.A = false end
		if k == Enum.KeyCode.D then freecamKeys.D = false end
		if k == Enum.KeyCode.Space then freecamKeys.Space = false end
		if k == Enum.KeyCode.LeftControl then freecamKeys.Ctrl = false end
	end)

	freecamInputChanged = UIS.InputChanged:Connect(function(input)
		if not freecamEnabled then return end
		if input.UserInputType == Enum.UserInputType.MouseMovement then
			freecamYaw -= input.Delta.X * freecamSens
			freecamPitch = math.clamp(freecamPitch - input.Delta.Y * freecamSens, -89, 89)
		end
	end)

	freecamConn = RS.RenderStepped:Connect(function(dt)
		if not freecamEnabled then return end

		local rotation = CFrame.Angles(0, math.rad(freecamYaw), 0) * CFrame.Angles(math.rad(freecamPitch), 0, 0)
		local move = Vector3.zero

		if freecamKeys.W then move += rotation.LookVector end
		if freecamKeys.S then move -= rotation.LookVector end
		if freecamKeys.A then move -= rotation.RightVector end
		if freecamKeys.D then move += rotation.RightVector end
		if freecamKeys.Space then move += Vector3.yAxis end
		if freecamKeys.Ctrl then move -= Vector3.yAxis end

		if move.Magnitude > 0 then
			freecamPos += move.Unit * freecamSpeed * dt
		end

		camera.CFrame = CFrame.new(freecamPos) * rotation
	end)
end

local function stopFreecam()
	if not freecamEnabled then return end
	freecamEnabled = false

	if freecamConn then freecamConn:Disconnect() freecamConn = nil end
	if freecamInputBegan then freecamInputBegan:Disconnect() freecamInputBegan = nil end
	if freecamInputEnded then freecamInputEnded:Disconnect() freecamInputEnded = nil end
	if freecamInputChanged then freecamInputChanged:Disconnect() freecamInputChanged = nil end

	UIS.MouseBehavior = Enum.MouseBehavior.Default
	UIS.MouseIconEnabled = true

	camera.CameraType = freecamOldType or Enum.CameraType.Custom
	camera.CameraSubject = freecamOldSubject

	unfreeze()
end

-- ;viewcam <tên> / ;unviewcam  (xem camera qua người chơi khác)
--============================================================

local viewingCam = false
local viewTarget = nil
local viewConn
local viewOldType, viewOldSubject

local function unviewcam()
	if not viewingCam then return end
	viewingCam = false
	viewTarget = nil

	if viewConn then
		viewConn:Disconnect()
		viewConn = nil
	end

	camera.CameraType = viewOldType or Enum.CameraType.Custom
	camera.CameraSubject = viewOldSubject
end

local function viewcam(target)
	if not target or target == player then return end

	local character = target.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	if not humanoid then return end

	if viewingCam then unviewcam() end

	viewingCam = true
	viewTarget = target
	viewOldType = camera.CameraType
	viewOldSubject = camera.CameraSubject

	camera.CameraType = Enum.CameraType.Custom
	camera.CameraSubject = humanoid

	viewConn = RS.RenderStepped:Connect(function()
		if not viewingCam then return end
		local char = viewTarget and viewTarget.Character
		local h = char and char:FindFirstChildOfClass("Humanoid")

		if not h or h.Health <= 0 then
			unviewcam()
			return
		end

		if camera.CameraSubject ~= h then
			camera.CameraSubject = h
		end
	end)
end

-- ;invisible / ;visible
-- Lưu ý: ghost chỉ tạo trên CLIENT của bạn (LocalScript), người
-- chơi khác KHÔNG thấy hiệu ứng này — chỉ ẩn nhân vật với chính bạn.
--============================================================

local invisible = false
local ghostClone
local savedCFrame
local savedTransparency = {}

local function createGhost(character)
	character.Archivable = true
	local clone = character:Clone()
	clone.Name = "ERROR422_Ghost"

	for _, obj in ipairs(clone:GetDescendants()) do
		if obj:IsA("BasePart") then
			obj.Transparency = 0.5
			obj.CanCollide = false
			obj.CanTouch = false
			obj.CanQuery = false
			obj.CastShadow = false
		elseif obj:IsA("Decal") or obj:IsA("Texture") then
			obj.Transparency = 0.5
		elseif obj:IsA("Script") or obj:IsA("LocalScript") then
			obj:Destroy()
		end
	end

	clone.Parent = workspace
	clone:PivotTo(savedCFrame)
	return clone
end

local function setInvisible()
	if invisible then return end
	local character, _, root = getCharacter()
	if not root then return end

	invisible = true
	savedCFrame = root.CFrame
	ghostClone = createGhost(character)

	for _, obj in ipairs(character:GetDescendants()) do
		if obj:IsA("BasePart") then
			savedTransparency[obj] = obj.LocalTransparencyModifier
			obj.LocalTransparencyModifier = 1
		elseif obj:IsA("Decal") or obj:IsA("Texture") then
			savedTransparency[obj] = obj.Transparency
			obj.Transparency = 1
		end
	end
end

local function setVisible()
	if not invisible then return end
	invisible = false

	if ghostClone then
		ghostClone:Destroy()
		ghostClone = nil
	end

	for obj, value in pairs(savedTransparency) do
		if obj and obj.Parent then
			if obj:IsA("BasePart") then
				obj.LocalTransparencyModifier = value
			elseif obj:IsA("Decal") or obj:IsA("Texture") then
				obj.Transparency = value
			end
		end
	end

	table.clear(savedTransparency)
end

local function tpToPlayer(name)
	local target = findPlayerByName(name)
	if not target then
		warn("Không tìm thấy player:", name)
		return
	end

	local _, _, root = getCharacter()
	local targetChar = target.Character
	local targetRoot = targetChar and targetChar:FindFirstChild("HumanoidRootPart")

	if root and targetRoot then
		root.CFrame = targetRoot.CFrame
	end
end

local ChatCommands = {
	[";fly"] = function() startFly() end,
	[";unfly"] = function() stopFly() end,
	[";flyspeed"] = function(_, args) setFlySpeed(args[2]) end,

	[";sit"] = function() sitDown() end,

	[";noclip"] = function() startNoclip() end,
	[";unnoclip"] = function() stopNoclip() end,

	[";speed"] = function(_, args) setSpeed(args[2]) end,
	[";unspeed"] = function() resetSpeed() end,

	[";jump"] = function() jumpNow() end,
	[";jumppower"] = function(_, args) setJumpPower(args[2]) end,
	[";unjumppower"] = function() resetJumpPower() end,

	[";gravity"] = function(_, args) setGravity(args[2]) end,
	[";ungravity"] = function() resetGravity() end,

	[";freeze"] = function() freeze() end,
	[";unfreeze"] = function() unfreeze() end,

	[";freecam"] = function() startFreecam() end,
	[";unfreecam"] = function() stopFreecam() end,

	[";viewcam"] = function(rest)
		local target = findPlayerByName(rest)
		if target then viewcam(target) end
	end,
	[";unviewcam"] = function() unviewcam() end,

	[";rejoin"] = function() rejoin() end,

	[";invisible"] = function() setInvisible() end,
	[";visible"] = function() setVisible() end,

	[";tp"] = function(_, args) tpToPlayer(args[2]) end,

	[";respawn"] = function() respawnSelf() end,
}

player.Chatted:Connect(function(message)
	local args = string.split(message, " ")
	local cmd = (args[1] or ""):lower()
	local rest = table.concat(args, " ", 2)

	local handler = ChatCommands[cmd]
	if handler then
		handler(rest, args)
	end
end)

local function applyBevel(frame, raised)
	local topLeft = raised and COLORS.Highlight or COLORS.Shadow
	local bottomRight = raised and COLORS.Shadow or COLORS.Highlight
	local topLeft2 = raised and COLORS.Face or COLORS.DarkShadow
	local bottomRight2 = raised and COLORS.DarkShadow or COLORS.Face

	local function line(pos, size, color)
		local l = Instance.new("Frame")
		l.BorderSizePixel = 0
		l.BackgroundColor3 = color
		l.Position = pos
		l.Size = size
		l.ZIndex = frame.ZIndex + 1
		l.Parent = frame
		return l
	end

	line(UDim2.new(0, 0, 0, 0), UDim2.new(1, 0, 0, 1), topLeft)
	line(UDim2.new(0, 0, 0, 0), UDim2.new(0, 1, 1, 0), topLeft)
	line(UDim2.new(0, 0, 1, -1), UDim2.new(1, 0, 0, 1), bottomRight)
	line(UDim2.new(1, -1, 0, 0), UDim2.new(0, 1, 1, 0), bottomRight)

	line(UDim2.new(0, 1, 0, 1), UDim2.new(1, -2, 0, 1), topLeft2)
	line(UDim2.new(0, 1, 0, 1), UDim2.new(0, 1, 1, -2), topLeft2)
	line(UDim2.new(0, 1, 1, -2), UDim2.new(1, -2, 0, 1), bottomRight2)
	line(UDim2.new(1, -2, 0, 1), UDim2.new(0, 1, 1, -2), bottomRight2)
end

--// KÉO THẢ CỬA SỔ
do
	local dragging = false
	local dragStart, startPos

	titleBar.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = window.Position

			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
				end
			end)
		end
	end)

	UIS.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch) then
			local delta = input.Position - dragStart
			window.Position = UDim2.new(
				startPos.X.Scale, startPos.X.Offset + delta.X,
				startPos.Y.Scale, startPos.Y.Offset + delta.Y
			)
		end
	end)
end

local function createRow(cmdName, index, action)
	local row = Instance.new("TextButton")
	row.Name = cmdName
	row.Size = UDim2.new(1, 0, 0, 30)
	row.BackgroundColor3 = index % 2 == 0 and Color3.fromRGB(240, 240, 240) or COLORS.FieldBg
	row.BorderSizePixel = 0
	row.AutoButtonColor = false
	row.Text = ""
	row.LayoutOrder = index
	row.Parent = list

	local icon = Instance.new("Frame")
	icon.Size = UDim2.fromOffset(18, 18)
	icon.Position = UDim2.fromOffset(6, 6)
	icon.BackgroundColor3 = COLORS.Accent
	icon.BorderSizePixel = 0
	icon.Parent = row
	applyBevel(icon, true)

	local label = Instance.new("TextLabel")
	label.BackgroundTransparency = 1
	label.Position = UDim2.fromOffset(32, 0)
	label.Size = UDim2.new(1, -40, 1, 0)
	label.Font = FONT
	label.Text = cmdName
	label.TextColor3 = COLORS.Text
	label.TextSize = 13
	label.TextXAlignment = Enum.TextXAlignment.Left
	label.Parent = row

	row.MouseEnter:Connect(function()
		row.BackgroundColor3 = Color3.fromRGB(10, 36, 106)
		label.TextColor3 = Color3.new(1, 1, 1)
	end)
	row.MouseLeave:Connect(function()
		row.BackgroundColor3 = index % 2 == 0 and Color3.fromRGB(240, 240, 240) or COLORS.FieldBg
		label.TextColor3 = COLORS.Text
	end)

	row.MouseButton1Click:Connect(function()
		if action then
			action()
		end
	end)

	return row
end

for i, cmd in ipairs(COMMANDS) do
	createRow(cmd.name, i, cmd.action)
end
